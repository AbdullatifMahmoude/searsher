سجل التعديلات (Audit Log)

المزامنة السحابية مع شيت جوجل

حفظ التعديلات والإضافات مباشرة في ملف شيت جوجل الأصلي

التطبيق يقرأ البيانات لحظياً ومباشرة من شيت جوجل المعتمد. لتفعيل الحفظ التلقائي في الشيت مباشرة عند تعديل أو إضافة أي عضو، يمكنك وضع رابط (Google Apps Script Web App) هنا:

كود Google Apps Script البسيط للشيت: 
function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var data = JSON.parse(e.postData.contents);
  if (data.action === "save") {
    var rows = sheet.getDataRange().getValues();
    var found = false;
    for (var i = 1; i < rows.length; i++) {
      if (rows[i][0] == data.record.id) {
        sheet.getRange(i + 1, 1, 1, 7).setValues([[
          data.record.id, data.record.name, data.record.job,
          data.record.salaryBefore, data.record.allowance,
          data.record.bonus10, data.record.salaryAfter
        ]]);
        found = true;
        break;
      }
    }
    if (!found) {
      sheet.appendRow([
        data.record.id, data.record.name, data.record.job,
        data.record.salaryBefore, data.record.allowance,
        data.record.bonus10, data.record.salaryAfter
      ]);
    }
  } else if (data.action === "delete") {
    var rows = sheet.getDataRange().getValues();
    for (var i = 1; i < rows.length; i++) {
      if (rows[i][0] == data.id) {
        sheet.deleteRow(i + 1);
        break;
      }
    }
  }
  return ContentService.createTextOutput(JSON.stringify({status: "ok"})).setMimeType(ContentService.MimeType.JSON);
}

2 الحفظ التلقائي للتدقيق والرمز عبر (Apps Script):

لتسجيل كل عملية تعديل في الشيت (سجل التدقيق) وتغيير الرمز من الموقع مباشرة: انشر الكود التالي كتطبيق ويب في الشيت:

كود Apps Script لشيت الأمان والتدقيق: 
function doPost(e) {
  try {
    var data = JSON.parse(e.postData.contents);
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    if (data.action === "set_pin") {
      var sheet = ss.getSheets()[0];
      sheet.getRange("A1").setValue("pin");
      sheet.getRange("B1").setValue("'" + (data.pin || ""));
      return ContentService.createTextOutput(JSON.stringify({status: "ok", pin: data.pin})).setMimeType(ContentService.MimeType.JSON);
    }
    if (data.action === "audit_log") {
      var auditSheet = ss.getSheetByName("سجل_التدقيق") || ss.insertSheet("سجل_التدقيق");
      if (auditSheet.getLastRow() === 0) {
        auditSheet.appendRow(["التاريخ والوقت", "نوع العملية", "المسؤول / العضو", "التفاصيل"]);
      }
      auditSheet.appendRow([data.timestamp, data.type, data.name, data.details]);
      return ContentService.createTextOutput(JSON.stringify({status: "ok"})).setMimeType(ContentService.MimeType.JSON);
    }
  } catch(err) {
    return ContentService.createTextOutput(JSON.stringify({status: "error", message: err.toString()})).setMimeType(ContentService.MimeType.JSON);
  }
}
